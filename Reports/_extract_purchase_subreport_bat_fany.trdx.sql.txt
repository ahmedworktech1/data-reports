WITH TrRep AS (&#xD;
    SELECT *,&#xD;
        ROW_NUMBER() OVER (&#xD;
            PARTITION BY ISNULL(NULLIF(LTRIM(RTRIM(FNCL_TR_NO)), ''), 'TR#' + CAST(TR_NO AS VARCHAR(20))), LOC, TR_TY&#xD;
            ORDER BY TR_DT DESC, TR_NO DESC&#xD;
        ) AS rn_tr&#xD;
    FROM AZTRH&#xD;
    WHERE (NULLIF(@Loc, '') IS NULL OR LOC = @Loc)&#xD;
      AND (@TrTy IS NULL OR TR_TY = @TrTy)&#xD;
      AND (NULLIF(@FnclTrNo, '') IS NULL OR FNCL_TR_NO = @FnclTrNo)&#xD;
),&#xD;
&#xD;
Ranked AS (&#xD;
    SELECT&#xD;
        ACC_NO,&#xD;
        FNCL_TR_NO,&#xD;
        TEC_STAT,&#xD;
        ROW_NUMBER() OVER (&#xD;
            PARTITION BY ACC_NO, FNCL_TR_NO&#xD;
            ORDER BY TR_DT DESC&#xD;
        ) AS rn&#xD;
    FROM AZTRH&#xD;
    WHERE ACC_NO IS NOT NULL&#xD;
      AND FNCL_TR_NO IS NOT NULL&#xD;
      AND (NULLIF(@Loc, '') IS NULL OR LOC = @Loc)&#xD;
      AND (@TrTy IS NULL OR TR_TY = @TrTy)&#xD;
      AND (NULLIF(@FnclTrNo, '') IS NULL OR FNCL_TR_NO = @FnclTrNo)&#xD;
),&#xD;
&#xD;
RankedDistinct AS (&#xD;
    SELECT DISTINCT&#xD;
        ACC_NO,&#xD;
        FNCL_TR_NO,&#xD;
        TEC_STAT&#xD;
    FROM Ranked&#xD;
    WHERE rn = 1&#xD;
),&#xD;
&#xD;
Suppliers AS (&#xD;
    SELECT&#xD;
        FNCL_TR_NO,&#xD;
        COUNT(CASE WHEN TEC_STAT = 0 THEN 1 END) AS عدد_العطاءات_المقبولة,&#xD;
        COUNT(CASE WHEN TEC_STAT = 1 THEN 1 END) AS عدد_العطاءات_المرفوضة,&#xD;
        COUNT(CASE WHEN TEC_STAT = 2 THEN 1 END) AS عدد_العطاءات_المعتذرة,&#xD;
        STRING_AGG(&#xD;
            CASE WHEN TEC_STAT = 0 THEN N'مورد رقم ' + CAST(ACC_NO AS NVARCHAR) END,&#xD;
            N' - '&#xD;
        ) AS مورد_مقبول,&#xD;
        STRING_AGG(&#xD;
            CASE WHEN TEC_STAT = 1 THEN N'مورد رقم ' + CAST(ACC_NO AS NVARCHAR) END,&#xD;
            N' - '&#xD;
        ) AS مورد_مرفوض,&#xD;
        STRING_AGG(&#xD;
            CASE WHEN TEC_STAT = 2 THEN N'مورد رقم ' + CAST(ACC_NO AS NVARCHAR) END,&#xD;
            N' - '&#xD;
        ) AS مورد_معتذر&#xD;
    FROM RankedDistinct&#xD;
    GROUP BY FNCL_TR_NO&#xD;
),&#xD;
&#xD;
AllMembers AS (&#xD;
    /* رئيس اللجنة */&#xD;
    SELECT DISTINCT&#xD;
        A.TR_NO,&#xD;
        A.PURCHAS_NO,&#xD;
        A.FNCL_TR_NO,&#xD;
        A.TR_DT,&#xD;
        A.TR_TY,&#xD;
        A.LOC,&#xD;
        A.Tchncl_Dcsn_Dt,&#xD;
        A.Fncl_Dcsn_Dt,&#xD;
        A.CMMTTE_RMRK_DCSN,&#xD;
        A.PYMNT_MTHD,&#xD;
        A.LoadType,&#xD;
        A.Vst_No,&#xD;
        A.Tr_Year,&#xD;
        A.Cmmtte_Dscn_No_Dcsn,&#xD;
        A.Cmmtte_Dscn_Dt_Dcsn,&#xD;
        A.Nt_Extrnl_Mmbr_Dcsn,&#xD;
        A.Order_No,&#xD;
        A.Tchncl_Opn_Dt,&#xD;
        A.PRCHS_OPRTN_TYP,&#xD;
        A.Tchncl_Dcsn_End_Dt,&#xD;
        A.rmrk9,&#xD;
        A.RMRK_EXTRA,&#xD;
        P.EMP_NAME AS MemberName,&#xD;
        N'رئيس اللجنة' AS MEMBER_TYPE,&#xD;
        NM.Emp_Name AS nm&#xD;
    FROM TrRep A&#xD;
    INNER JOIN PYEMPMF_H P&#xD;
        ON P.EMP_NO = TRY_CAST(&#xD;
            REPLACE(REPLACE(REPLACE(COALESCE(A.Cmmtte_Chrmn_Dcsn,''), '''',''),'"',''),' ','')&#xD;
        AS INT)&#xD;
    LEFT JOIN PYEMPMF_H NM ON NM.EMP_NO = A.ENTEREDBY&#xD;
    WHERE A.rn_tr = 1&#xD;
&#xD;
    UNION ALL&#xD;
&#xD;
    /* عضو داخلي */&#xD;
    SELECT DISTINCT&#xD;
        A.TR_NO,&#xD;
        A.PURCHAS_NO,&#xD;
        A.FNCL_TR_NO,&#xD;
        A.TR_DT,&#xD;
        A.TR_TY,&#xD;
        A.LOC,&#xD;
        A.Tchncl_Dcsn_Dt,&#xD;
        A.Fncl_Dcsn_Dt,&#xD;
        A.CMMTTE_RMRK_DCSN,&#xD;
        A.PYMNT_MTHD,&#xD;
        A.LoadType,&#xD;
        A.Vst_No,&#xD;
        A.Tr_Year,&#xD;
        A.Cmmtte_Dscn_No_Dcsn,&#xD;
        A.Cmmtte_Dscn_Dt_Dcsn,&#xD;
        A.Nt_Extrnl_Mmbr_Dcsn,&#xD;
        A.Order_No,&#xD;
        A.Tchncl_Opn_Dt,&#xD;
        A.PRCHS_OPRTN_TYP,&#xD;
        A.Tchncl_Dcsn_End_Dt,&#xD;
        A.rmrk9,&#xD;
        A.RMRK_EXTRA,&#xD;
        P.EMP_NAME AS MemberName,&#xD;
        N'عضو داخلي' AS MEMBER_TYPE,&#xD;
        NM.Emp_Name AS nm&#xD;
    FROM TrRep A&#xD;
    OUTER APPLY (&#xD;
        SELECT DISTINCT TRY_CAST(value AS INT) AS EMP_NO&#xD;
        FROM STRING_SPLIT(&#xD;
            REPLACE(REPLACE(REPLACE(COALESCE(A.Cmmtte_Intrnl_Mmbrs_Dcsn,''), ';',','),'''',''),'"',''),&#xD;
            ','&#xD;
        )&#xD;
        WHERE TRY_CAST(value AS INT) IS NOT NULL&#xD;
    ) X&#xD;
    INNER JOIN PYEMPMF_H P ON P.EMP_NO = X.EMP_NO&#xD;
    LEFT JOIN PYEMPMF_H NM ON NM.EMP_NO = A.ENTEREDBY&#xD;
    WHERE A.rn_tr = 1&#xD;
&#xD;
    UNION ALL&#xD;
&#xD;
    /* عضو خارجي */&#xD;
    SELECT DISTINCT&#xD;
        A.TR_NO,&#xD;
        A.PURCHAS_NO,&#xD;
        A.FNCL_TR_NO,&#xD;
        A.TR_DT,&#xD;
        A.TR_TY,&#xD;
        A.LOC,&#xD;
        A.Tchncl_Dcsn_Dt,&#xD;
        A.Fncl_Dcsn_Dt,&#xD;
        A.CMMTTE_RMRK_DCSN,&#xD;
        A.PYMNT_MTHD,&#xD;
        A.LoadType,&#xD;
        A.Vst_No,&#xD;
        A.Tr_Year,&#xD;
        A.Cmmtte_Dscn_No_Dcsn,&#xD;
        A.Cmmtte_Dscn_Dt_Dcsn,&#xD;
        A.Nt_Extrnl_Mmbr_Dcsn,&#xD;
        A.Order_No,&#xD;
        A.Tchncl_Opn_Dt,&#xD;
        A.PRCHS_OPRTN_TYP,&#xD;
        A.Tchncl_Dcsn_End_Dt,&#xD;
        A.rmrk9,&#xD;
        A.RMRK_EXTRA,&#xD;
        L.DSCR_AR AS MemberName,&#xD;
        N'عضو خارجي' AS MEMBER_TYPE,&#xD;
        NM.Emp_Name AS nm&#xD;
    FROM TrRep A&#xD;
    OUTER APPLY (&#xD;
        SELECT DISTINCT TRY_CAST(value AS INT) AS CODE&#xD;
        FROM STRING_SPLIT(&#xD;
            REPLACE(REPLACE(REPLACE(COALESCE(A.Cmmtte_Extrnl_Mmbrs_Dcsn,''), ';',','),'''',''),'"',''),&#xD;
            ','&#xD;
        )&#xD;
        WHERE TRY_CAST(value AS INT) IS NOT NULL&#xD;
    ) X&#xD;
    INNER JOIN AZLISTS L ON L.CODE = X.CODE AND L.REC_TYPE = 61&#xD;
    LEFT JOIN PYEMPMF_H NM ON NM.EMP_NO = A.ENTEREDBY&#xD;
    WHERE A.rn_tr = 1&#xD;
&#xD;
    UNION ALL&#xD;
&#xD;
    /* عضو أمن داخلي */&#xD;
    SELECT DISTINCT&#xD;
        A.TR_NO,&#xD;
        A.PURCHAS_NO,&#xD;
        A.FNCL_TR_NO,&#xD;
        A.TR_DT,&#xD;
        A.TR_TY,&#xD;
        A.LOC,&#xD;
        A.Tchncl_Dcsn_Dt,&#xD;
        A.Fncl_Dcsn_Dt,&#xD;
        A.CMMTTE_RMRK_DCSN,&#xD;
        A.PYMNT_MTHD,&#xD;
        A.LoadType,&#xD;
        A.Vst_No,&#xD;
        A.Tr_Year,&#xD;
        A.Cmmtte_Dscn_No_Dcsn,&#xD;
        A.Cmmtte_Dscn_Dt_Dcsn,&#xD;
        A.Nt_Extrnl_Mmbr_Dcsn,&#xD;
        A.Order_No,&#xD;
        A.Tchncl_Opn_Dt,&#xD;
        A.PRCHS_OPRTN_TYP,&#xD;
        A.Tchncl_Dcsn_End_Dt,&#xD;
        A.rmrk9,&#xD;
        A.RMRK_EXTRA,&#xD;
        P.EMP_NAME AS MemberName,&#xD;
        N'عضو أمن داخلي' AS MEMBER_TYPE,&#xD;
        NM.Emp_Name AS nm&#xD;
    FROM TrRep A&#xD;
    INNER JOIN PYEMPMF_H P&#xD;
        ON P.EMP_NO = TRY_CAST(&#xD;
            REPLACE(REPLACE(REPLACE(COALESCE(A.Cmmtte_Intrnl_Scrty_Dcsn,''), '''',''),'"',''),' ','')&#xD;
        AS INT)&#xD;
    LEFT JOIN PYEMPMF_H NM ON NM.EMP_NO = A.ENTEREDBY&#xD;
    WHERE A.rn_tr = 1&#xD;
&#xD;
    UNION ALL&#xD;
&#xD;
    /* عضو مجلس دولة */&#xD;
    SELECT DISTINCT&#xD;
        A.TR_NO,&#xD;
        A.PURCHAS_NO,&#xD;
        A.FNCL_TR_NO,&#xD;
        A.TR_DT,&#xD;
        A.TR_TY,&#xD;
        A.LOC,&#xD;
        A.Tchncl_Dcsn_Dt,&#xD;
        A.Fncl_Dcsn_Dt,&#xD;
        A.CMMTTE_RMRK_DCSN,&#xD;
        A.PYMNT_MTHD,&#xD;
        A.LoadType,&#xD;
        A.Vst_No,&#xD;
        A.Tr_Year,&#xD;
        A.Cmmtte_Dscn_No_Dcsn,&#xD;
        A.Cmmtte_Dscn_Dt_Dcsn,&#xD;
        A.Nt_Extrnl_Mmbr_Dcsn,&#xD;
        A.Order_No,&#xD;
        A.Tchncl_Opn_Dt,&#xD;
        A.PRCHS_OPRTN_TYP,&#xD;
        A.Tchncl_Dcsn_End_Dt,&#xD;
        A.rmrk9,&#xD;
        A.RMRK_EXTRA,&#xD;
        L.DSCR_AR AS MemberName,&#xD;
        N'عضو مجلس دولة' AS MEMBER_TYPE,&#xD;
        NM.Emp_Name AS nm&#xD;
    FROM TrRep A&#xD;
    INNER JOIN AZLISTS L&#xD;
        ON L.CODE = TRY_CAST(&#xD;
            REPLACE(REPLACE(REPLACE(COALESCE(A.Nt_Extrnl_Mmbr_Dcsn,''), '''',''),'"',''),' ','')&#xD;
        AS INT)&#xD;
       AND L.REC_TYPE = 61&#xD;
    LEFT JOIN PYEMPMF_H NM ON NM.EMP_NO = A.ENTEREDBY&#xD;
    WHERE A.rn_tr = 1&#xD;
)&#xD;
&#xD;
SELECT DISTINCT&#xD;
    N.TR_NO,&#xD;
    N.PURCHAS_NO,&#xD;
    N.FNCL_TR_NO,&#xD;
    S.عدد_العطاءات_المقبولة,&#xD;
    S.عدد_العطاءات_المرفوضة,&#xD;
    S.عدد_العطاءات_المعتذرة,&#xD;
    S.مورد_مقبول,&#xD;
    S.مورد_مرفوض,&#xD;
    S.مورد_معتذر,&#xD;
    N.TR_DT,&#xD;
    N.Tchncl_Dcsn_Dt,&#xD;
    N.Fncl_Dcsn_Dt,&#xD;
    N.CMMTTE_RMRK_DCSN,&#xD;
    N.Nt_Extrnl_Mmbr_Dcsn,&#xD;
    SC.DSCR_AR AS Nt_Extrnl_Mmbr_Dcsn_Name,&#xD;
    N.TR_TY,&#xD;
    N.LOC,&#xD;
    N.MEMBER_TYPE,&#xD;
    N.MemberName,&#xD;
    CASE WHEN N.MEMBER_TYPE = N'رئيس اللجنة' THEN 0 ELSE 1 END AS SORT_ORDER,&#xD;
    N.nm,&#xD;
    T.DSCR_AR AS OprTypeName,&#xD;
    PM.DSCR_AR AS PYMNT_MTHD_Name,&#xD;
    LT.DSCR_AR AS LoadTypeName,&#xD;
    N.Vst_No,&#xD;
    N.Tr_Year,&#xD;
    N.Cmmtte_Dscn_No_Dcsn,&#xD;
    N.Cmmtte_Dscn_Dt_Dcsn,&#xD;
    N.Order_No,&#xD;
    N.Tchncl_Opn_Dt,&#xD;
    N.Tchncl_Dcsn_End_Dt,&#xD;
    N.rmrk9,&#xD;
    N.PRCHS_OPRTN_TYP,&#xD;
    PT.DSCR_AR AS PRCHS_OPRTN_TYP_Name&#xD;
FROM AllMembers N&#xD;
LEFT JOIN Suppliers S ON S.FNCL_TR_NO = N.FNCL_TR_NO&#xD;
LEFT JOIN AZTRTY T ON T.TR_TY = N.TR_TY&#xD;
LEFT JOIN AZTRTY PT ON PT.TR_TY = N.PRCHS_OPRTN_TYP&#xD;
LEFT JOIN AppCodes PM ON PM.ID = '11031' AND PM.SEQ = N.PYMNT_MTHD&#xD;
LEFT JOIN AppCodes LT ON LT.ID = '10244' AND LT.SEQ = N.LoadType&#xD;
LEFT JOIN AZLISTS SC&#xD;
    ON SC.CODE = TRY_CAST(&#xD;
        REPLACE(REPLACE(REPLACE(COALESCE(N.Nt_Extrnl_Mmbr_Dcsn,''), '''',''),'"',''),' ','')&#xD;
    AS INT)&#xD;
   AND SC.REC_TYPE = 61&#xD;
ORDER BY&#xD;
    N.TR_NO,&#xD;
    SORT_ORDER,&#xD;
    N.MemberName;
